cmake_minimum_required(VERSION 3.24)
project(xMenuForgeContextMenuPoc LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This proof of concept must be built on Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(xMenuForgeContextMenu SHARED ContextMenu.cpp)
target_compile_definitions(xMenuForgeContextMenu PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(xMenuForgeContextMenu PRIVATE ole32 shell32 shlwapi)
set_target_properties(xMenuForgeContextMenu PROPERTIES OUTPUT_NAME "xMenuForgeContextMenu")
add_executable(xMenuForgePocHost WIN32 PocHost.cpp)
target_compile_definitions(xMenuForgePocHost PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(xMenuForgePocHost PRIVATE shell32)
install(TARGETS xMenuForgeContextMenu xMenuForgePocHost RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION lib)
